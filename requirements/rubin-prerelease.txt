# Public prerelease PyTorch nightly pins for Rubin bring-up.
# Installed when INSTALL_RUBIN_PRERELEASE=true (see docker/Dockerfile).
# torch is the cu135 nightly, matching the Rubin image's CUDA 13.5. PyTorch
# publishes no cu135 torchvision or torchaudio yet, so they use the cu134
# nightlies of the same date through CUDA minor-version compatibility.
# TODO(rubin-bringup): delete once Rubin APIs ship in regular PyTorch wheels
# (vllm-project/vllm#49735).
--extra-index-url https://download.pytorch.org/whl/nightly/cu135
--extra-index-url https://download.pytorch.org/whl/nightly/cu134

# Use the same nightly pins for x86_64 and aarch64.
# torch pulls triton==3.9.0+gitb7ddcf86 from PyTorch's index. Rubin images set
# TRITON_PTXAS_BLACKWELL_PATH to the ptxas-blackwell (13.4) it bundles, and the
# image build fails if Triton would assemble SM107 with ptxas 13.5.37, which
# silently miscompiles some sm_107a code. Triton main bundles only 13.5.37.
# Before bumping torch, check cmake/nvidia-toolchain-version.json at the Triton
# commit in torch's triton==X+git<sha> requirement.
torch==2.16.0.dev20261116+cu135
torchvision==0.30.0.dev20261116+cu134
torchaudio==2.11.0.dev20261116+cu134

cuda-python==13.5.0
cuda-bindings==13.5.0
nixl==1.5.2
