# Common dependencies
-r common.txt

numba == 0.65.0 # Required for N-gram speculative decoding

# Dependencies for NVIDIA GPUs
#
# torch 2.15.0 (a release candidate) and torchaudio 2.11.0 are published for
# cu132 only on the test channel. Without it the resolver falls back to PyPI,
# whose torchaudio is a cu130 build that aborts every CUDA job at import:
#   RuntimeError: Detected that PyTorch and TorchAudio were compiled with
#   different CUDA versions. PyTorch has CUDA version 13.2 whereas TorchAudio
#   has CUDA version 13.0.
# Drop this once torch 2.15.0 and cu132 torchaudio are on the stable index.
--extra-index-url https://download.pytorch.org/whl/test/cu132
torch==2.15.0
torchaudio==2.11.0
# These must be updated alongside torch
torchvision==0.29.1 # Required for phi3v processor. See https://github.com/pytorch/vision?tab=readme-ov-file#installation for corresponding version
torchcodec >= 0.14
PyNvVideoCodec==2.0.4
flashinfer-python==0.7.2
apache-tvm-ffi==0.1.11
tilelang==0.1.12
# >=1.31.0 keeps execution plans alive for the CUDA graphs recorded from them
# (NVIDIA/cudnn-frontend#1195); older versions can replay an unloaded kernel.
nvidia-cudnn-frontend>=1.31.0
# Required for LLM_NVTX_SCOPES_FOR_PROFILING=1
nvtx==0.2.15
# Required for faster safetensors model loading
fastsafetensors >= 0.3.3
instanttensor >= 0.2.2

# QuACK and Cutlass DSL for FA4 (cute-DSL implementation)
nvidia-cutlass-dsl[cu13]==4.8.0
quack-kernels==0.6.5  # Required by MSA

# Tokenspeed_MLA for faster mla with spec decode
tokenspeed-mla==2.0.3; platform_system == "Linux"

# Humming kernels for quantization gemm
humming-kernels[cu13]==0.1.17
